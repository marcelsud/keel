A practical reference

Keel SDK components

Keel brings APIs, data storage, events, and background jobs into one development workflow. You can build a service that manages a product catalog, tracks categories and inventory, and updates an index in the background using a consistent set of tools in TypeScript or Go. Keel connects your services, starts the local infrastructure, and restarts your service as you edit, so you spend less time on setup and more time building your application's behavior. Start with one service and add more as your application grows.

src/service.ts · create and store a product
import { createKeel, ValidationError } from "@marcelsud/keel.ts";

const app = await createKeel({
  name: "catalog",
  url: process.env.KEEL_URL,
  token: process.env.KEEL_TOKEN,
});

const createProduct = app.action({
  name: "createProduct",
  args: app.schema.object({
    name: app.schema.string(),
    category: app.schema.string(),
    priceCents: app.schema.integer(),
    stock: app.schema.integer(),
    active: app.schema.boolean(),
  }),
  async handler(ctx, input) {
    if (!input.name.trim() || !input.category.trim() ||
        input.priceCents < 0 || input.priceCents > 2_147_483_647 ||
        input.stock < 0 || input.stock > 2_147_483_647) {
      throw new ValidationError("invalid product");
    }
    const id = await ctx.collection.insert("products", input);
    return { id, ...input };
  },
});

await app.start();
console.log("catalog ready");
main.go · create and store a product
package main

import (
  "context"
  "encoding/json"
  "fmt"
  "log"
  "math"
  "os"
  "os/signal"
  "strings"
  "syscall"

  "github.com/marcelsud/keel/sdk"
)

func main() {
  app, err := sdk.Connect(sdk.Options{
    Name: "catalog", URL: os.Getenv("KEEL_URL"),
    Token: os.Getenv("KEEL_TOKEN"),
  })
  if err != nil { log.Print(err); return }
  defer app.Stop()

  err = app.Action("createProduct", func(
    ctx *sdk.Context, raw json.RawMessage,
  ) (any, error) {
    var input map[string]any
    if json.Unmarshal(raw, &input) != nil {
      return nil, &sdk.Error{Code: 400, Message: "invalid product"}
    }
    name, _ := input["name"].(string)
    category, _ := input["category"].(string)
    priceCents, priceOK := input["priceCents"].(float64)
    stock, stockOK := input["stock"].(float64)
    active, activeOK := input["active"].(bool)
    if strings.TrimSpace(name) == "" || strings.TrimSpace(category) == "" ||
      !priceOK || math.Trunc(priceCents) != priceCents ||
      priceCents < 0 || priceCents > 2_147_483_647 ||
      !stockOK || math.Trunc(stock) != stock || stock < 0 || stock > 2_147_483_647 ||
      !activeOK {
      return nil, &sdk.Error{Code: 400, Message: "invalid product"}
    }
    product := map[string]any{
      "name": name, "category": category, "priceCents": priceCents,
      "stock": stock, "active": active,
    }
    id, err := ctx.Collection.Insert(ctx, "products", product)
    if err != nil { return nil, err }
    product["id"] = id
    return product, nil
  })
  if err != nil { log.Print(err); return }

  ctx, stop := signal.NotifyContext(
    context.Background(), os.Interrupt, syscall.SIGTERM,
  )
  defer stop()
  if err := app.Start(ctx); err != nil { log.Print(err); return }
  fmt.Println("catalog ready")
  <-ctx.Done()
}
src/client.ts · create a product and print the result
import { createKeel } from "@marcelsud/keel.ts";

const app = await createKeel({
  name: "catalog-client",
  url: process.env.KEEL_URL,
  token: process.env.KEEL_TOKEN,
});

try {
  const product = await app.call<{
    id: string; name: string; category: string;
    priceCents: number; stock: number; active: boolean;
  }>("catalog", "createProduct", {
    name: "Trail Backpack", category: "outdoor",
    priceCents: 7990, stock: 12, active: true,
  });
  console.log(`${product.name} | ${product.category} | ${product.priceCents} cents | stock ${product.stock} | active ${product.active}`);
} finally {
  await app.stop();
}
cmd/client/main.go · create a product and print the result
package main

import (
  "context"
  "fmt"
  "log"
  "os"

  "github.com/marcelsud/keel/sdk"
)

func main() {
  app, err := sdk.Connect(sdk.Options{
    Name: "catalog-client", URL: os.Getenv("KEEL_URL"),
    Token: os.Getenv("KEEL_TOKEN"),
  })
  if err != nil { log.Print(err); return }
  defer app.Stop()
  ctx := context.Background()

  var product struct {
    ID, Name, Category string
    PriceCents, Stock int32
    Active bool
  }
  err = app.Call(ctx, "catalog", "createProduct", map[string]any{
    "name": "Trail Backpack", "category": "outdoor",
    "priceCents": 7990, "stock": 12, "active": true,
  }, &product)
  if err != nil { log.Print(err); return }
  fmt.Printf("%s | %s | %d cents | stock %d | active %t\n",
    product.Name, product.Category, product.PriceCents, product.Stock, product.Active,
  )
}

Start the service in the first terminal. Keel runs the local environment and restarts your service when its source changes:

Terminal 1 · start the catalog
keel dev

no protobuf contracts; skipping generation is expected for this JSON example. Wait for catalog ready, then open a second terminal in the same directory and run the client:

Terminal 2 · create a product
bun run src/client.ts
Terminal 2 · create a product
go run ./cmd/client
Expected output · both languages
Trail Backpack | outdoor | 7990 cents | stock 12 | active true

The product is stored in the products collection with a generated ID. Every client run creates another product. Prices use integer cents; price and stock must be whole numbers from 0 to 2,147,483,647. Both implementations reject invalid input with a 400 service error and keep the same product fields.

Run bun run check to type-check the service and client. The input schema validates JSON types; the handler checks the catalog rules. A generic client result annotation does not validate the response at runtime.

Run go build ./... to check the service and client. The handler validates the same JSON types and catalog rules before saving. The client decodes the response into a Go struct.

The TypeScript and Go tabs change all examples on this page. Each reference example below demonstrates the same operation with the same product fields in both languages; use each scenario independently. Add registrations before starting the app. In the lifecycle example, insert them at the marked position.

New projects use keel.toml; existing keel.json files remain supported. Keep one configuration file and one keel dev process per project. Ctrl-C stops the service and local environment.

CLI v0.2.2+ also includes embedded SQL. Follow the SQL setup for migrations and service credentials. SQL callback transactions cover one service's database; collection writes, events, and queue operations commit independently.

both keel.toml and keel.jsonChoose one configuration file and remove the other. Keel reports an ambiguity error when both exist.
Use a different port or share an environment with another project

For a separate project with its own environment, start on a different port:

Terminal 1 · separate project, separate environment
keel dev --port 4223

Set the same URL when calling it from the second terminal:

Terminal 2 · client on port 4223
KEEL_URL=nats://127.0.0.1:4223 bun run src/client.ts
Terminal 2 · client on port 4223
KEEL_URL=nats://127.0.0.1:4223 go run ./cmd/client

keel dev --port -1 chooses a free port. Copy the URL from keel ready into the client's KEEL_URL.

To share an already running environment instead, run this from another service project:

Another project · reuse the environment on port 4222
KEEL_URL=nats://127.0.0.1:4222 keel dev

KEEL_URL makes Keel connect your service to that environment. Each service still has its own name and startup command.

Handler context and errors

Action, subscription, and worker handlers receive the SDK building blocks. HTTP handlers receive the request, parameters, query, and body; call an action to reuse service logic.

Inside a handler

ctx.kv, ctx.collection, ctx.pubsub, ctx.queue, ctx.client(), ctx.call(), and ctx.log are available.

ctx.KV, ctx.Collection, ctx.Pubsub, ctx.Queue, ctx.Call(), and ctx.Log are available. Generated typed clients also accept the handler context.

Service errors

Throw ValidationError for a 400 response or ServiceError(code, message) for an explicit service error. Clients receive a ServiceError. Unexpected handler errors return 500.

Return &sdk.Error{Code: 400, Message: "..."} for an explicit service error. Callers can inspect it with errors.As. Unexpected handler errors return 500.

Requests and shutdown

Requests default to a 5-second timeout, configurable with timeout. Await app.stop() to drain active handlers and deliveries.

Pass a context.Context to calls. Requests use a 5-second timeout if the context has no deadline. app.Stop() drains active handlers and deliveries.

What is available today

Typed unary RPC, data operations, events, workers, and version coexistence are implemented. GitHub contract installation, dated EOL notices, and the alpha / beta / stable lifecycle are planned.
